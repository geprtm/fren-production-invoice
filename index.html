            function setZoom(scale) {

                currentZoom = Math.max(0.4, Math.min(1.3, scale));

                const container = document.getElementById('paper-scale-container');

                container.style.transform = `scale(${currentZoom})`;

                document.getElementById('zoom-level-text').textContent = Math.round(currentZoom * 100) + '%';

            }


            function fitPreviewToViewport() {

                const panel = document.getElementById('right-preview-panel');

                if (!panel) return;

                const availableWidth = panel.clientWidth - 48;

                const paperWidth = 794;

                

                if (availableWidth > 0 && availableWidth < paperWidth) {

                    setZoom(availableWidth / paperWidth);

                } else {

                    setZoom(1);

                }

            }


            window.switchTab = function (tabInput) {

                const raw = String(tabInput || '1');

                const tabId = raw.includes('2') ? '2' : '1';


                document.querySelectorAll('.tab-switch-btn').forEach((b) => {

                    const isTarget = b.getAttribute('data-tab') === tabId;

                    if (isTarget) {

                        b.classList.add('bg-white', 'text-slate-900', 'border-slate-200');

                        b.classList.remove('text-slate-500', 'border-transparent');

                    } else {

                        b.classList.remove('bg-white', 'text-slate-900', 'border-slate-200');

                        b.classList.add('text-slate-500', 'border-transparent');

                    }

                });


                const panel1 = document.getElementById('tab-panel-1');

                const panel2 = document.getElementById('tab-panel-2');

                if (panel1) panel1.classList.toggle('hidden', tabId !== '1');

                if (panel2) panel2.classList.toggle('hidden', tabId !== '2');

            };


            document.querySelectorAll('.tab-switch-btn').forEach((btn) => {

                btn.addEventListener('click', () => {

                    window.switchTab(btn.getAttribute('data-tab'));

                });

            });

                if (!desc) return;

            document.getElementById('editor-items-list').addEventListener('click', (e) => {

                const btn = e.target.closest('button[data-action]');

                if (!btn) return;

                const action = btn.getAttribute('data-action');

                const idx = parseInt(btn.getAttribute('data-index'), 10);


                if (action === 'up' && idx > 0) {

                    const temp = state.items[idx - 1];

                    state.items[idx - 1] = state.items[idx];

                    state.items[idx] = temp;

                    if (editingItemIndex === idx) editingItemIndex = idx - 1;

                    renderAll();

                } else if (action === 'down' && idx < state.items.length - 1) {

                    const temp = state.items[idx + 1];

                    state.items[idx + 1] = state.items[idx];

                    state.items[idx] = temp;

                    if (editingItemIndex === idx) editingItemIndex = idx + 1;

                    renderAll();

                } else if (action === 'edit') {

                    const item = state.items[idx];

                    editingItemIndex = idx;

                    document.getElementById('input-desc').value = item.desc;

                    document.getElementById('input-qty').value = item.qty;

                    document.getElementById('input-unit').value = item.unit;

                    document.getElementById('input-price').value = item.price;

                    document.getElementById('item-form-heading').textContent = 'Edit Item #' + (idx + 1);

                    document.getElementById('item-edit-badge').classList.remove('hidden');

                    document.getElementById('btn-submit-item').textContent = 'Simpan Perubahan';

                    document.getElementById('btn-cancel-edit').classList.remove('hidden');

                    updateFormSubtotalPreview();

                    renderEditorItemsList();

                    document.getElementById('input-desc').focus();

                } else if (action === 'delete') {

                    state.items.splice(idx, 1);

                    if (editingItemIndex === idx) {

                        resetItemForm();

                    } else if (editingItemIndex > idx) {

                        editingItemIndex--;

                    }

                    renderAll();

                }

            });
